# The budget planner takes its whole state from the query string, so every
# island page links a pre-filled variant (?i=naxos%3A4 ...). They all canonical
# back to the bare /trip-cost/ page, so crawling ~200 of them is pure waste.
# The planner itself stays fully crawlable.
Disallow: /trip-cost/?i=
Disallow: /el/trip-cost/?i=

# Cloudflare's email-obfuscation rewrites the mailto: on /privacy/ into a
# /cdn-cgi/ decoder link. It isn't a page — Bing's crawler was following it
# and logging a 4xx. Keep the obfuscation (it protects the address), just
# stop crawlers walking into it.
Disallow: /cdn-cgi/
